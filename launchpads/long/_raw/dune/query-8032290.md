-- LONG widget: total volume per stock token (spec #2)

-- volume_usd = ALL stock-token flow through LONG pools (incl. protocol buyback legs)

-- user_volume_usd = trader-initiated flow only

WITH s AS(

SELECT*FROM query_8032229

)

SELECT

numeraire_symbol AS stock,

SUM(amount_usd)AS volume_usd,

 100.0 *SUM(amount_usd)/SUM(SUM(amount_usd))OVER()AS share_pct,

SUM(CASE WHEN NOT is_buyback THEN amount_usd ELSE 0 END)AS user_volume_usd,

SUM(CASE WHEN block_time>=NOW()-INTERVAL'24'HOUR THEN amount_usd ELSE 0 END)AS volume_24h_usd,

COUNT(*)AS swaps,

COUNT(DISTINCT CASE WHEN NOT is_buyback THEN trader END)AS traders,

COUNT(DISTINCT asset)AS tokens_traded

FROM s

GROUP BY 1

ORDER BY 2 DESC
