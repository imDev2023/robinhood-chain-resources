SELECT f.feed,

COALESCE(

regexp_extract(regexp_extract(from_utf8(ct.code),'[A-Za-z0-9 .&\-]{2,25} ?/ ?USD'),'Robinhood ([A-Z]+) ?/ ?USD', 1),
