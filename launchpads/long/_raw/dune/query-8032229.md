-- LONG foundation 4/4: long_swaps

-- Every swap in a LONG pool (v4 bonding + graduated v2/v3), with the numeraire-side

-- amount, buyback flag (protocol buyback legs: sender = fee manager — these must be

-- excluded from USER volume but kept for pool-balance accounting), USD value from

-- Chainlink hourly prices, and the tx sender as trader identity.

WITH pools AS(

SELECT*FROM query_8032178

),

prices AS(

-- q3 (Chainlink aggregators) + q3b (on-chain print-derived prices for feedless

-- stocks, incl. RCAT/AMC/GLXY). AI- and memecoin-quoted pairs have NO price row,

-- so the inner price join below excludes them from the dashboard BY DESIGN

-- (scope: real stocks only - Nate 2026-08-20).

SELECT*FROM query_8032188

UNION ALL

SELECT*FROM query_8391616

),

v4_swaps AS(

SELECT

l.block_time,l.block_date,l.tx_hash,l.index AS evt_index,

varbinary_substring(l.topic1, 13, 20)AS sender,

l.topic3 AS pool_id,

varbinary_to_int256(varbinary_substring(l.data, 97, 32))AS amount0,

varbinary_to_int256(varbinary_substring(l.data, 129, 32))AS amount1

FROM robinhood.logs l

WHERE l.topic0= 0x1d9f7b5e406d8c887155e1a78e070d2d41c5d0444dab8b21612f846835c27183

AND l.contract_address= 0x4e3468951d49f2eea976ed0d6e75ffcb44a9a544

AND l.block_date>=DATE'2026-07-01'

),

SELECT

s.block_time,s.block_date,s.tx_hash,s.evt_index,

p.asset,p.numeraire,p.ticker,

'v4'AS venue,

(s.sender= 0x6f02324d20cc679d0e585290caa6b16bacbc0f77)AS is_buyback,

ABS(CASE WHEN p.asset_is_currency0 THEN s.amount1 ELSE s.amount0 END)AS numeraire_amount_raw,
