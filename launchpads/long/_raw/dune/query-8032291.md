-- LONG widget: top assets table — volume_usd counts ALL stock flow through the asset's

-- pools (incl. protocol buybacks); user_volume_usd is trader flow only

WITH s AS(

SELECT*FROM query_8032229

),

la AS(

SELECT asset,MIN(launch_time)AS launch_time FROM query_8032167 GROUP BY 1

)

SELECT

get_href('https://app.long.xyz/tokens/0x'||LOWER(TO_HEX(s.asset)),COALESCE(MAX(s.ticker),'0x'||LOWER(TO_HEX(s.asset))))AS token,

MAX(s.numeraire_symbol)AS stock,

MIN(la.launch_time)AS launched,

SUM(s.amount_usd)AS volume_usd,

SUM(CASE WHEN NOT s.is_buyback THEN s.amount_usd ELSE 0 END)AS user_volume_usd,

SUM(CASE WHEN s.block_time>=NOW()-INTERVAL'24'HOUR THEN s.amount_usd ELSE 0 END)AS volume_24h_usd,

SUM(CASE WHEN s.block_time>=NOW()-INTERVAL'7'DAY THEN s.amount_usd ELSE 0 END)AS volume_7d_usd,

COUNT(CASE WHEN NOT s.is_buyback THEN 1 END)AS user_trades,

COUNT(DISTINCT CASE WHEN NOT s.is_buyback THEN s.trader END)AS traders

FROM s

JOIN la ON la.asset=s.asset

GROUP BY s.asset

ORDER BY volume_usd DESC

LIMIT 50
